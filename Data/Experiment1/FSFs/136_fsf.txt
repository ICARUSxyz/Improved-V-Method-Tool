    f = null && ex = NullArgument
    ||
    min > max && ex = BadSequence
    ||
yMin = 0 && ex = None && r = min
    ||
and yMax = 0 && ex = None && r = max
    ||
and yStart = 0 && ex = None && r = start
    ||
and yStart ≠ 0 && ex = None
        && r ∈ [min, max]   // proceed from start with Müller iterations
    ||
xNew - xPrev <= tol     && ex = None
       && r = xNew && |f(r)| <= fAcc
    ||
xProj - xPrev <= tol      && ex = None
       && r = xProj && |f(r)| <= fAcc
    ||
xSafe - xPrev <= tol
      && ex = None
       && r = xSafe && |f(r)| <= fAcc
    ||
    evaluations > maxEval && ex = TooManyEvaluations
