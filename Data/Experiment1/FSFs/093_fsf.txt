        currentTemperature < minTemperature && heatingActive = true, output "Low temperature detected: " + currentTemperature + "°C. Heating activated.", Heating activated.
        ||
        currentTemperature > maxTemperature && heatingActive = false, output "High temperature detected: " + currentTemperature + "°C. Heating deactivated.", Heating deactivated.
        ||
        currentTemperature <= maxTemperature && heatingActive = false, output "Temperature optimal: " + currentTemperature + "°C. Heating remains off.", Heating remains off.
