        currentTemperature <= maxTemperature && currentTemperature = currentTemperature + changeAmount, coolingActive = false, heatingActive = false, Temperature maintained within optimal range.
        ||
        currentTemperature > maxTemperature && coolingActive = true, heatingActive = false, Cooling system activated.
        ||
        currentTemperature < minTemperature && coolingActive = false, heatingActive = true, Heating system activated.
