module SmartElevatorOverloadDetection
    var currentLoad, maxLoadCapacity: real
    process updateLoad(newLoad: real) returns (currentLoad: real)
        functional scenario form:
        currentLoad > maxLoadCapacity && output "ALERT: Overload detected (" + currentLoad + " kg). Elevator cannot operate.", Overload alarm triggered, elevator operation blocked.
        ||
        currentLoad <= maxLoadCapacity && output "Current load: " + currentLoad + " kg. Elevator operating normally.", Elevator operates normally.
    end process
end module
