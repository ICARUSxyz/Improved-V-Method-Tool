module SmartVendingMachineInventory
    var currentInventory, totalInventory: integer 
    var pricePerItem, totalCost: real 
    process purchaseItems(quantity: integer) returns (totalCost: real)
        functional scenario form:
        quantity <= currentInventory && totalCost = quantity * pricePerItem, currentInventory = currentInventory – quantity, Purchase successful, totalCost = quantity * pricePerItem, currentInventory updated.
        ||
        quantity > currentInventory && totalCost = 0, Purchase failed, totalCost = 0, currentInventory unchanged.
    end process
end module
