module OnlineShoppingDiscount
    var totalAmount, discount, finalPrice: real 
    var promoCode: string
    var purchaseLog: sequence of string 
    process calculateDiscount(totalAmount: real, promoCode: string) discount: real
        functional scenario form:
        totalAmount > 200 && discount = totalAmount * 0.10, 10% discount applied.
        ||
        totalAmount > 100 && discount = totalAmount * 0.05, 5% discount applied.
        ||
        promoCode = 20 && discount = discount + 20, Additional $20 discount applied.
        ||
        promoCode = 10 && discount = discount + 10, Additional $10 discount applied.
        ||
        discount > totalAmount && discount = totalAmount, Discount capped at total purchase amount.
    end process
end module
