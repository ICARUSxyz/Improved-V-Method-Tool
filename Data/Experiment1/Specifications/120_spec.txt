module WindTurbineMonitoringEx
    var currentPower, maxPower, minPower: real 
    var powerLog: sequence of string 
    process updatePower(changeAmount: real)
        functional scenario form:
        currentPower >= minPower && powerLog = powerLog ⨁ ("Power Updated: " + currentPower + " kW"), Power output updated successfully.
        ||
        currentPower > maxPower && powerLog = powerLog ⨁ ("Overload Warning: " + currentPower + " kW, Emergency Shutdown Recommended."), Overload warning triggered, shutdown recommended.
        ||
        currentPower < minPower && powerLog = powerLog ⨁ ("Low Efficiency Alert: " + currentPower + " kW, Maintenance Recommended."), Low efficiency alert triggered, maintenance recommended.
    end process
end module
