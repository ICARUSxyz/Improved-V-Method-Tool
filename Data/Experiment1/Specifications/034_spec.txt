module BatteryChargingManagement
    var currentCharge, maxCharge, minCharge: real 
    var chargeLog: sequence of string
    process updateCharge(changeAmount: real)
        functional scenario form:
        currentCharge >= minCharge && chargeLog = chargeLog ⨁ ("Charge Level Updated: " + currentCharge + "%"), Charge level updated successfully.
        ||
        currentCharge > maxCharge && chargeLog = chargeLog ⨁ ("Overcharge Warning: " + currentCharge + "%, Stop Charging Immediately."), Overcharge warning triggered, stop charging.
        ||
        currentCharge < minCharge && chargeLog = chargeLog ⨁ ("Low Battery Alert: " + currentCharge + "%, Immediate Charging Recommended."), Low battery alert triggered, immediate charging recommended.
    end process
end module
