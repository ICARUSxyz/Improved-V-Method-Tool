module SmartWaterPressureControl
    var currentPressure, minPressure, maxPressure: real
    var pumpActive, releaseValveActive: boolean
    process updatePressure(newPressure: real) returns (currentPressure: real, pumpActive: boolean, releaseValveActive: boolean)
        functional scenario form:
        currentPressure < minPressure && pumpActive = true, releaseValveActive = false, output "ALERT: Water pressure too low (" + currentPressure + " bar). Activating water pump.", Water pump activated.
        ||
        currentPressure > maxPressure && pumpActive = false, releaseValveActive = true, output "ALERT: Water pressure too high (" + currentPressure + " bar). Activating pressure release valve.", Pressure release valve activated.
        ||
        currentPressure >= minPressure && pumpActive = false, releaseValveActive = false, output "Water pressure optimal (" + currentPressure + " bar). No action required.", No control action required.
    end process
end module
