module SmartWaterAllocation
    var totalWaterSupply: real 
    var waterRequests: sequence of (string, real) 
    var allocationLog: sequence of string
    process allocateWater(waterRequests: sequence of (string, real))
        functional scenario form:
        totalWaterRequests <= totalWaterSupply && totalWaterSupply = totalWaterSupply - sum(waterRequests)and allocationLog = allocationLog ⨁ "Water allocated successfully", Water allocated successfully.
        ||
        totalWaterRequests > totalWaterSupply && totalWaterSupply = 0and allocationLog = allocationLog ⨁ "Water rationing applied", Water supply insufficient, rationing applied.
    end process
end module
