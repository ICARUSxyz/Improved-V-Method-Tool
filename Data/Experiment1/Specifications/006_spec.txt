module BankTransferSystem
    var senderBalance, recipientBalance, transferAmount: real 
    var transactionLog: sequence of string
    process transfer(senderBalance: real, recipientBalance: real, transferAmount: real) flag: boolean
        functional scenario form:
        transferAmount <= senderBalance && flag = true and senderBalance = senderBalance - transferAmountand recipientBalance = recipientBalance + transferAmount and transactionLog = transactionLog ⨁ ("Transfer recorded"), Valid transfer, amount deducted and added to recipient, transaction logged
        ||
        transferAmount > senderBalance && flag = false, Invalid transfer, insufficient funds, values unchanged
        ||
        transferAmount <= 0 && flag = false, Invalid transfer, amount must be positive, values unchanged
    end process
end module
