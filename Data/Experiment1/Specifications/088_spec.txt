module SmartRefrigeratorDefrostControl
    var currentFrostThickness, defrostThreshold: real
    var defrostCycleActive: boolean 
    process updateFrostThickness(newThickness: real) returns (currentFrostThickness: real, defrostCycleActive: boolean)
        functional scenario form:
            currentFrostThickness > defrostThreshold && defrostCycleActive = true, output "Frost thickness high: " + currentFrostThickness + " mm. Defrost cycle activated.", Defrost cycle activated.
            ||
            currentFrostThickness <= defrostThreshold && defrostCycleActive = false, output "Frost thickness normal: " + currentFrostThickness + " mm. No defrost needed.", No defrost cycle triggered.
    end process
end module
