module SmartSolarPanelCleaning
    var currentDustLevel, dustThreshold: real
    process updateDustLevel(newDustLevel: real) returns (currentDustLevel: real)
        functional scenario form:
        currentDustLevel > dustThreshold && output "ALERT: Dust level (" + currentDustLevel + "%) exceeds threshold (" + dustThreshold + "%). Cleaning cycle activated.", Cleaning cycle activated.
        ||
        currentDustLevel <= dustThreshold && output "Dust level (" + currentDustLevel + "%) is within acceptable limits. No cleaning required.", No cleaning action required.
    end process
end module
