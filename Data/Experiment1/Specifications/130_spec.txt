module SmartLiquidLevelControlEx
    var currentLiquidLevel, maxLiquidLevel, minLiquidLevel: real  
    var refillActive: boolean 
    process updateLiquidLevel(changeAmount: real)
        functional scenario form:
        currentLiquidLevel >= minLiquidLevel && Liquid level maintained.
        ||
        currentLiquidLevel < minLiquidLevel && refillActive = true, Refill system activated.
        ||
        currentLiquidLevel > maxLiquidLevel && refillActive = false, Refill system deactivated.
    end process
end module
