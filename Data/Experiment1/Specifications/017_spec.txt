module SmartTemperatureControl
    var currentTemperature, minTemp, maxTemp: real 
    var temperatureLog: sequence of string 
    process updateTemperature(newTemp: real) flag: boolean
        functional scenario form:
        newTemp < minTemp && flag = true and temperatureLog = temperatureLog ⨁ ("Heating system activated."), Temperature too low! Heating activated.
        ||
        newTemp > maxTemp && flag = true and temperatureLog = temperatureLog ⨁ ("Cooling system activated."), Temperature too high! Cooling activated.
      ||
   newTemp <= maxTemp && Temperature stable, no action required.
    end process
end module
